let API_BASE = localStorage.getItem("soccer_api_url") || "https://soccer-intelligence-engine.onrender.com"; function changeApiUrl() { const url = prompt("Enter Render backend URL:", API_BASE); if (url) { API_BASE = url.replace(/\/$/, ""); localStorage.setItem("soccer_api_url", API_BASE); checkHealth(); } } async function apiFetch(path, opts = {}) { const normalizedPath = path.startsWith("/") ? path : `/${path}`; const endpoints = []; // 1. Try configured Render deployment URL if (API_BASE) { endpoints.push(`${API_BASE}${normalizedPath}`); } // 2. Seamless local proxy failover (/api/* and root) const localTarget = normalizedPath.startsWith("/api") ? normalizedPath : `/api${normalizedPath}`; if (!endpoints.includes(localTarget)) endpoints.push(localTarget); if (!endpoints.includes(normalizedPath)) endpoints.push(normalizedPath); let lastError = null; for (const target of endpoints) { try { const r = await fetch(target, { headers: { "Content-Type": "application/json" }, ...opts }); if (r.ok) return await r.json(); let detail = `API error ${r.status}`; try { const errJson = await r.json(); if (errJson.detail) detail = errJson.detail; } catch (e) {} lastError = new Error(detail); } catch (err) { lastError = err; } } throw lastError || new Error("Failed to communicate with prediction engine"); }
